# Parse LLM output into a typed model, don't regex it

LLMs work far better with structured contracts than post-hoc string parsing.

When you extract data using regex, you have to handle missing matches, unexpected formatting, and brittle fallbacks.

By passing a strict Pydantic model directly to structured-output APIs (e.g. native tool-calling frameworks or schema enforcement libraries), the LLM is forced to comply with your schema at generation time.

```python
import re
from typing import Literal

from pydantic import BaseModel

# ❌ Brittle: What happens when the regex misses or formats change?
def extract_sentiment(text: str) -> str:
    response = llm.complete(f"Analyse sentiment: {text}")
    match = re.search(r"\b(positive|negative|neutral)\b", response, re.IGNORECASE)
    return match.group(1).lower() if match else "unknown"

# ✅ Clean: Schema-enforced outputs with typed validation
class SentimentResult(BaseModel):
    sentiment: Literal["positive", "negative", "neutral"]
    confidence: float

def extract_sentiment(text: str) -> SentimentResult:
    return llm.complete(
        f"Analyse sentiment: {text}",
        response_model=SentimentResult
    )
```
